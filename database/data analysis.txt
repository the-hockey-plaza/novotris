/* last users with more than 5 games */
select u.id, 
  u.name, 
  (select count(*) from game g where g.user_id = u.id) as nr_games
from nov_user u
where (select count(*) from game g where g.user_id = u.id) > 5
order by u.id desc

/* last cancelled games */
select u.id, u.name, g.beginn, g.level, g.timestamp_live, g.score_live
from nov_user u, game g
where g.user_id = u.id
  and g.score is null
  order by g.id desc
  
/* levels per user */
select max(g.level) as lvl, 
  (select count(*) from game g1 where g1.user_id = u.id) as cnt,   
  u.id
from nov_user u, game g
where g.user_id = u.id
  and g.ende is not null
group by u.id
order by u.id desc

/* highscore per level per user */
select u.id,
  (select count(*) from game g where g.user_id = u.id) as nr_games,
  (select max(g1.score) from game g1 where g1.user_id = u.id and g1.level = 1) as high1,
  (select max(g2.score) from game g2 where g2.user_id = u.id and g2.level = 2) as high2,
  (select max(g3.score) from game g3 where g3.user_id = u.id and g3.level = 3) as high3,
  (select max(g4.score) from game g4 where g4.user_id = u.id and g4.level = 4) as high4
from nov_user u
where (select max(g1.score) from game g1 where g1.user_id = u.id and g1.level = 1) > 1
order by u.id desc
