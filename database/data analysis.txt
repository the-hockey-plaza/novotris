/* last users with more than 5 games */
select u.id, 
  u.name, 
  (select count(*) from game g where g.user_id = u.id) as nr_games
from nov_user u
where (select count(*) from game g where g.user_id = u.id) > 5
order by u.id desc

/* last cancelled games */
select u.id, u.name, g.beginn, g.level, g.timestamp_live, g.score_live
from nov_user u, game g
where g.user_id = u.id
  and g.score is null
  order by g.id desc
  
/* max level of users active in last 30 days */
select u.id, 
  u.name, 
  u.mobile,
  u.level,
  (select count(*) 
   from game g 
   where g.user_id = u.id
     and g.beginn >=  CURDATE() - INTERVAL 30 DAY) as nr_games,
  (select max(g2.beginn)
   from game g2
	  where g2.user_id = u.id 
    and g2.score is not null) last_game,
  (select max(g3.score)
   from game g3
	  where g3.user_id = u.id 
    and g3.score is not null
    and g3.level = 1) max_score_1    
from nov_user u
where exists (
  select g1.id
  from game g1 
  where g1.user_id = u.id 
    and g1.score is not null
    and g1.beginn >=  CURDATE() - INTERVAL 30 DAY)

/* last games of a single user */
select g.*
from game g
where g.user_id = 2663
order by g.id desc
